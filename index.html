// --- GEMINI PROMPT DE EXTRAÇÃO COM TRATAMENTO AMIGÁVEL DE ERRO --- async function callGeminiExtract(payloadPart) { if (!navigator.onLine) { throw new Error('Sem conexão com a internet. Verifique sua rede.'); } if (!currentApiKey) { toggleConfigModal(); throw new Error('Configure sua API Key primeiro.'); } const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3.6-flash:generateContent?key=${currentApiKey}`; const systemPrompt = `Você é um extrator contábil. Extraia as informações financeiras da entrada e retorne APENAS um objeto JSON com: - description: breve resumo do gasto ou ganho - amount: número positivo (float) - type: "income" para receitas/ganhos ou "expense" para compras/gastos/saídas - category: escolha EXATAMENTE uma das seguintes opções: * "Contas & Moradia": energia elétrica, água, internet, gás, aluguel, condomínio, IPTU. * "Alimentação": supermercado, restaurante, lanches, delivery, padaria, feira. * "Transporte": combustível, uber, táxi, passagem, pedágio, manutenção de carro. * "Saúde & Beleza": farmácia, remédios, consultas, dentista, academia, corte de cabelo. * "Lazer & Assinaturas": netflix, spotify, cinema, passeios, bares, viagens. * "Compras & Pessoal": roupas, sapatos, eletrônicos, presentes, itens pessoais. * "Educação": cursos, livros, mensalidade escolar/faculdade. * "Ganhos / Renda": salário, freelance, pix recebido, vendas, reembolsos. * "Investimentos": aportes, poupança, tesouro, ações. * "Outros": qualquer outro gasto que não pertença a nenhuma acima. Se não conseguir identificar valor, retorne amount como 0.`; const response = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [{ parts: [{ text: systemPrompt }, payloadPart] }], generationConfig: { response_mime_type: "application/json" } }) }); if (!response.ok) { const err = await response.json().catch(() => ({})); const rawMsg = err.error?.message || ''; // Tradução e simplificação inteligente de limites e erros if (response.status === 429 || rawMsg.includes('Quota exceeded') || rawMsg.includes('quota')) { const retryMatch = rawMsg.match(/retry in ([\d\.]+)s/i); if (retryMatch) { const seconds = Math.ceil(parseFloat(retryMatch[1])); throw new Error(`Limite temporário de requisições. Aguarde ${seconds} segundos para tentar de novo.`); } throw new Error('Limite momentâneo atingido. Aguarde cerca de 30 segundos.'); } else if (response.status === 400 || rawMsg.includes('API_KEY_INVALID') || rawMsg.includes('key not valid')) { throw new Error('Chave de API inválida. Verifique em ⚙️ Chave.'); } throw new Error('Não foi possível conectar com a IA no momento. Tente novamente.'); } const result = await response.json(); return JSON.parse(result.candidates[0].content.parts[0].text); }
